    function clearAdvancedSearch() {

        if (originalIcon) {
          iconElement.text(originalIcon);
          iconElement.removeClass('qf-loading');
          buttonElement.removeData('original-icon');
        }
      });

  function toggleTechnicalDetails(button) {
    const card = button.closest('.terminal-card');
    const detailsSection = card.querySelector('.terminal-details');
    const toggleIcon = button.querySelector('.toggle-icon');

    // Toggle da visibilidade
    if (detailsSection.classList.contains('show')) {
      detailsSection.classList.remove('show');
      toggleIcon.classList.remove('expanded');
    } else {
      detailsSection.classList.add('show');
      toggleIcon.classList.add('expanded');
    }
  }

  // Garantir que todas as seções estejam fechadas inicialmente
  document.addEventListener('DOMContentLoaded', function() {
    const allDetails = document.querySelectorAll('.terminal-details');
    allDetails.forEach(details => {
      details.classList.remove('show');
    });

    const allToggleIcons = document.querySelectorAll('.toggle-icon');
    allToggleIcons.forEach(icon => {
      icon.classList.remove('expanded');
    });
  });

<script>
  function toggleTechnicalDetails(button) {
    const card = button.closest('.terminal-card');
    const detailsSection = card.querySelector('.terminal-details');
    const toggleIcon = button.querySelector('.toggle-icon');

    // Toggle da visibilidade
    if (detailsSection.classList.contains('show')) {
      detailsSection.classList.remove('show');
      toggleIcon.classList.remove('expanded');
    } else {
      detailsSection.classList.add('show');
      toggleIcon.classList.add('expanded');
    }
  }

  // Garantir que todas as seções estejam fechadas inicialmente
  document.addEventListener('DOMContentLoaded', function() {
    const allDetails = document.querySelectorAll('.terminal-details');
    allDetails.forEach(details => {
      details.classList.remove('show');
    });

    const allToggleIcons = document.querySelectorAll('.toggle-icon');
    allToggleIcons.forEach(icon => {
      icon.classList.remove('expanded');
    });
  });

<script>
  function toggleTechnicalDetails(button) {
    const card = button.closest('.terminal-card');
    const detailsSection = card.querySelector('.terminal-details');
    const toggleIcon = button.querySelector('.toggle-icon');

    // Toggle da visibilidade
    if (detailsSection.classList.contains('show')) {
      detailsSection.classList.remove('show');
      toggleIcon.classList.remove('expanded');
    } else {
      detailsSection.classList.add('show');
      toggleIcon.classList.add('expanded');
    }
  }

  // Garantir que todas as seções estejam fechadas inicialmente

    });

    allToggleIcons.forEach(icon => {
      icon.classList.remove('expanded');
    });
  });

<script type="text/javascript">
(function() {
  const STORAGE_KEY = 'terminal-view-mode';
  const DEFAULT_VIEW = 'cards';
  
  // Função para obter o modo de visualização do localStorage
  function getViewMode() {
    try {
      const stored = localStorage.getItem(STORAGE_KEY);
      return stored === 'list' ? 'list' : DEFAULT_VIEW;
    } catch (e) {
      return DEFAULT_VIEW;
    }
  }
  
  // Função para salvar o modo de visualização no localStorage
  function saveViewMode(mode) {
    try {
      localStorage.setItem(STORAGE_KEY, mode);
    } catch (e) {
      console.warn('Failed to save view mode:', e);
    }
  }
  
  // Função para alternar o modo de visualização
  function switchViewMode(mode) {
    // Salvar preferência
    saveViewMode(mode);
    
    // Atualizar URL e recarregar página
    const url = new URL(window.location.href);
    url.searchParams.set('viewMode', mode);
    
    // Recarregar a página com o novo modo
    window.location.href = url.toString();
  }
  
  // Inicializar ao carregar a página
  $(document).ready(function() {
    const urlParams = new URLSearchParams(window.location.search);
    const urlMode = urlParams.get('viewMode');
    const currentMode = urlMode || document.querySelector('#terminal-grid')?.getAttribute('data-view-mode') || DEFAULT_VIEW;
    const savedMode = getViewMode();
    
    // Se não houver parâmetro na URL mas houver modo salvo, aplicar o modo salvo
    if (!urlMode && savedMode !== currentMode) {
      switchViewMode(savedMode);
      return; // A função já fará o reload
    }
    
    // Atualizar os botões baseado no modo atual
    document.querySelectorAll('.terminal-view-toggle .toggle-button').forEach(btn => {
      if (btn.getAttribute('data-view') === currentMode) {
        btn.classList.add('active');
      } else {
        btn.classList.remove('active');
      }
    });
    
    // Salvar o modo atual se for diferente do salvo
    if (currentMode !== savedMode) {
      saveViewMode(currentMode);
    }
    
    // Adicionar event listeners aos botões
    document.querySelectorAll('.terminal-view-toggle .toggle-button').forEach(btn => {
      btn.addEventListener('click', function(e) {
        e.preventDefault();
        const viewMode = this.getAttribute('data-view');
        if (viewMode && viewMode !== currentMode) {
          switchViewMode(viewMode);
        }
      });
    });
  });
})();
